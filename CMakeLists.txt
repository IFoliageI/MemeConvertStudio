cmake_minimum_required(VERSION 3.20)
project(MemeStudio LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "MemeStudio requires Windows 10 or newer.")
endif()
add_executable(MemeStudio WIN32 src/main.cpp src/gui.cpp src/util.cpp src/image.cpp src/api.cpp src/job.cpp src/plugins.cpp tests/tests.cpp)
target_compile_features(MemeStudio PRIVATE cxx_std_20)
target_compile_definitions(MemeStudio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(MemeStudio PRIVATE user32 gdi32 msimg32 windowscodecs ole32 oleaut32 uuid comctl32 comdlg32 shell32 shlwapi winhttp crypt32 bcrypt propsys)
if(MSVC)
    target_compile_options(MemeStudio PRIVATE /utf-8 /W4 /EHsc)
    target_link_options(MemeStudio PRIVATE /ENTRY:wmainCRTStartup "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
    set_property(TARGET MemeStudio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(MemeStudioExample SHARED plugins/example/plugin.cpp)
target_compile_features(MemeStudioExample PRIVATE cxx_std_20)
set_target_properties(MemeStudioExample PROPERTIES PREFIX "" OUTPUT_NAME "example")
if(MSVC)
    target_compile_options(MemeStudioExample PRIVATE /utf-8 /W4 /EHsc)
    set_property(TARGET MemeStudioExample PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_dependencies(MemeStudio MemeStudioExample)
add_custom_command(TARGET MemeStudio POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:MemeStudio>/plugins/example"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:MemeStudioExample>" "$<TARGET_FILE_DIR:MemeStudio>/plugins/example/example.dll"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/plugins/example/plugin.json" "$<TARGET_FILE_DIR:MemeStudio>/plugins/example/plugin.json")
enable_testing()
add_test(NAME studio-self-test COMMAND MemeStudio --self-test "${CMAKE_CURRENT_BINARY_DIR}/test-output")
